# ------------------------------------------------------------ 触发：剑拔弩张
country_event = {
	id = AOR_GER_weimar_congress.8
	title = AOR_GER_weimar_congress.8.t
	desc = AOR_GER_weimar_congress.8.d
	picture = GFX_report_event_generic_strike
	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = AOR_GER_weimar_congress.8.a
		custom_effect_tooltip = AOR_GER_weimar_congress.8.tt
		hidden_effect = {
			# 三大焦点的胜利数决定工会归属：3 → 左，1-2 → 中，0 → 右
			set_variable = { AOR_union_focus_wins = 0 }
			if = {
				limit = { has_country_flag = AOR_union_focus_1_won }
				add_to_variable = { AOR_union_focus_wins = 1 }
			}
			if = {
				limit = { has_country_flag = AOR_union_focus_2_won }
				add_to_variable = { AOR_union_focus_wins = 1 }
			}
			if = {
				limit = { has_country_flag = AOR_union_focus_3_won }
				add_to_variable = { AOR_union_focus_wins = 1 }
			}
			set_country_flag = AOR_union_congress_settled
			if = {
				limit = { check_variable = { AOR_union_focus_wins > 2 } }
				country_event = { id = AOR_GER_weimar_congress.9 }
			}
			else_if = {
				limit = { check_variable = { AOR_union_focus_wins > 0 } }
				country_event = { id = AOR_GER_weimar_congress.10 }
			}
			else = {
				country_event = { id = AOR_GER_weimar_congress.11 }
			}
		}
	}
}
