#-------------------------------------------------- 1920.8.11 移交国际委员会
country_event = {
	id = AOR_GER_versailles.7
	title = AOR_GER_versailles.7.t
	desc = AOR_GER_versailles.7.d
	picture = GFX_news_event_generic_sign_treaty2
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		set_country_flag = AOR_GER_schleswig_handover
		set_global_flag = AOR_GER_schleswig_handover

		# 北区 839 与弗伦斯堡 1043 移交国际委员会
		release = NSH
		if = {
			limit = { NOT = { NSH = { owns_state = 839 } } }
			NSH = { transfer_state = 839 }
		}
		if = {
			limit = { NOT = { NSH = { owns_state = 1043 } } }
			NSH = { transfer_state = 1043 }
		}
		if = {
			limit = { NOT = { NSH = { owns_state = 1125 } } }
			NSH = { transfer_state = 1125 }
		}
		NSH = {
			add_state_core = 839
			add_state_core = 1043
			add_state_core = 1125
		}

		# 国际委员会的担保方
		if = {
			limit = { ENG = { exists = yes } }
			ENG = { give_guarantee = NSH }
		}
		if = {
			limit = { FRA = { exists = yes } }
			FRA = { give_guarantee = NSH }
		}
	}

	option = {
		name = AOR_GER_versailles.7.a
	}
}
