			if = {
				limit = {
					has_character = GER_02_02_Robert_Dittmann

				deactivate_advisor = GER_02_02_Robert_Dittmann

				deactivate_advisor = GER_02_03_Theodor_Leipart

				limit = {

				deactivate_advisor = GER_Richard_Mueller_regular_cabinet

	}
	option = {
		name = AOR_GER_union_competition.1.a
		custom_effect_tooltip = AOR_GER_union_competition.1.effect_tt
		hidden_effect = { news_event = { id = AOR_news_europe.60 } }
	}

country_event = {
    id = AOR_GER_union_competition.5
    title = AOR_GER_union_competition.5.t
    desc = AOR_GER_union_competition.5.desc
    is_triggered_only = yes
    fire_only_once = yes
    immediate = {
        clr_country_flag = GER_SPD_union_reform_open
        hidden_effect = {
            remove_decision = GER_SPD_union_mobilize
            remove_decision = GER_SPD_union_coordinate
        }
        set_country_flag = GER_SPD_union_congress_started
        add_to_variable = { GER_SPD_gui_revision = 1 }
        set_variable = { GER_union_focus_victories = 0 }
        if = { limit = { has_country_flag = GER_SPD_union_industrial_done } add_to_variable = { GER_union_focus_victories = 1 } }
        if = { limit = { has_country_flag = GER_SPD_union_militant_done } add_to_variable = { GER_union_focus_victories = 1 } }
        if = {
            limit = { OR = { has_country_flag = GER_SPD_union_soviet_independence_done has_country_flag = GER_SPD_union_soviet_integration_done } }
            add_to_variable = { GER_union_focus_victories = 1 }
        }
    }
    option = {
        name = AOR_GER_union_competition.5.a
        hidden_effect = {
            if = { limit = { check_variable = { GER_union_focus_victories > 2 } } country_event = { id = AOR_GER_union_competition.8 } }
            else_if = { limit = { check_variable = { GER_union_focus_victories > 0 } } country_event = { id = AOR_GER_union_competition.7 } }
            else = { country_event = { id = AOR_GER_union_competition.6 } }
        }
    }
}
