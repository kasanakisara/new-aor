# ============================================================
# GER_SPD_decisions —— 粮食系统 Mission / 决议组
# 策划案：《德国决议组设计》粮食部分（AOR 模组）
#
# 【变量约定】（国家级变量，建议在 GER_now_what / 开局事件里 set_variable 初始化）
#   GER_food_surplus                粮食余量        （0~100+，越高越好）
#   GER_agri_development            农业发展度      （0~100+，参照 TNO 做 buff/debuff，越高越好）
#   GER_peasant_resentment          农民反感度      （0~100，越低越好）
#                                     · 影响征粮 / 粮食税成功概率
#                                     · 过低(<10) 降 SPD 支持度；过高(>90) 增 SPD 支持度
#   GER_SPD_support                 SPD 支持度      （分地 +，合作社 -）
#
# 【基调 = country flag】（二选一互斥，由 GER_land_distribution / GER_cooperative_reform 设置）
#   GER_agri_tone_land_distributed   分地：+SPD 支持度，但征粮/粮食税成功率与收益下降（护田）
#   GER_agri_tone_cooperative        合作社：-SPD 支持度，但征粮/粮食税成功率与收益上升（组织化）
#   三个 mission 通过 has_country_flag 检测基调，输出不同结果
#
# 【地区分类】敌方核心/摇摆区 / 控制区 / SPD核心区 尚未定稿，本文件先写通用逻辑，
#   等地区策划出来后再通过 state_flag / tag 分流各 mission 的结算。
# ============================================================

GER_SPD_decisions = {

    # =====================================================
    # 1) 征粮  GER_grain_requisition_mission
    #    适用：敌方核心区 / 摇摆区
    #    效果：增加反感度；反感度影响成功概率；降低农业发展度
    # =====================================================
    GER_grain_requisition_mission = {
        allowed = {
            tag = GER
        }
        available = {
            always = no
        }
        activation = {
            always = no
        }
        # 反感度爆表立即取消，防止暴动前硬征
        cancel_trigger = {
            check_variable = { GER_peasant_resentment > 95 }
        }
        is_good = no
        days_mission_timeout = 30
        selectable_mission = no
        timeout_effect = {
            # === 成功率判定：基础 60%，农民反感度越高成功率越低 ===
            #   反感 ≤50：60%   |   50~75：约 36%   |   >75：约 18%
            #   基调修正：合作社 x1.2（组织化好征），分地 x0.8（护田难征）
            set_variable = { GER_req_success_flag = 0 }
            random = {
                chance = 0.6
                modifier = {
                    factor = 0.6
                    check_variable = { GER_peasant_resentment > 50 }
                }
                modifier = {
                    factor = 0.5
                    check_variable = { GER_peasant_resentment > 75 }
                }
                modifier = {
                    factor = 1.2
                    has_country_flag = GER_agri_tone_cooperative
                }
                modifier = {
                    factor = 0.8
                    has_country_flag = GER_agri_tone_land_distributed
                }
                effect = {
                    set_variable = { GER_req_success_flag = 1 }
                }
            }
            if = {
                limit = {
                    check_variable = { GER_req_success_flag = 1 }
                }
                # 成功：粮到手，农民更反感，农业发展受损
                add_to_variable = { GER_food_surplus = 10 }
                add_to_variable = { GER_peasant_resentment = 5 }
                add_to_variable = { GER_agri_development = -2 }
                # 基调结果：合作社组织化 → 收上来的粮更多
                if = {
                    limit = { has_country_flag = GER_agri_tone_cooperative }
                    add_to_variable = { GER_food_surplus = 5 }
                }
                # 分地 → 农民留粮自用，征到的更少
                if = {
                    limit = { has_country_flag = GER_agri_tone_land_distributed }
                    add_to_variable = { GER_food_surplus = -3 }
                }
                # TODO: 成功事件（可给地区加"抗粮"延迟惩罚）
            }
            else = {
                # 失败：没征上来，农民照样记恨，地也踩坏了
                add_to_variable = { GER_peasant_resentment = 3 }
                add_to_variable = { GER_agri_development = -1 }
                # TODO: 失败小事件 / 舆论
            }
            # 轮转：结算完重新挂起，等待再次激活
            activate_mission = GER_grain_requisition_mission
        }
    }

                    limit = { has_country_flag = GER_agri_tone_cooperative }
                    add_to_variable = { GER_food_surplus = 5 }
                }
                # 分地 → 农民留粮自用，征到的更少
                if = {
                    limit = { has_country_flag = GER_agri_tone_land_distributed }
                    add_to_variable = { GER_food_surplus = -3 }
                }
            }
            else = {
                # 失败：没征上来，农民照样记恨，地也踩坏了
                add_to_variable = { GER_peasant_resentment_westphalia = 3 }
                add_to_variable = { GER_agri_development = -1 }
            }
            # 轮转：结算完重新挂起，等待再次激活
            activate_mission = GER_grain_requisition_mission_westphalia
        }
    }

    # =====================================================
    # 粮食税（威斯特伐利亚）  GER_grain_tax_mission_westphalia
    #    适用：控制区 / SPD核心区（制度化征收，不主动加反感）
    #    效果：反感度影响成功概率；降低农业发展度
    # =====================================================
    GER_grain_tax_mission_westphalia = {
        allowed = {
            tag = GER
        }
        available = {
            always = no
        }
        activation = {
            always = no
        }
        cancel_trigger = {
            check_variable = { GER_peasant_resentment_westphalia > 95 }
        }
        is_good = no
        days_mission_timeout = 30
        selectable_mission = no
        timeout_effect = {
            # 成功率：基础 70%，该地农民反感度越高成功率越低
            set_variable = { GER_tax_success_flag_westphalia = 0 }
            random = {
                chance = 0.7
                modifier = {
                    factor = 0.6
                    check_variable = { GER_peasant_resentment_westphalia > 60 }
                }
                modifier = {
                    factor = 0.5
                    check_variable = { GER_peasant_resentment_westphalia > 80 }
                }
                modifier = {
                    factor = 1.2
                    has_country_flag = GER_agri_tone_cooperative
                }
                modifier = {
                    factor = 0.8
                    has_country_flag = GER_agri_tone_land_distributed
                }
                effect = {
                    set_variable = { GER_tax_success_flag_westphalia = 1 }
                }
            }
            if = {
                limit = {
                    check_variable = { GER_tax_success_flag_westphalia = 1 }
                }
                # 成功：稳定入账，农业略受损
                add_to_variable = { GER_food_surplus = 8 }
                add_to_variable = { GER_agri_development = -1 }
                # 合作社 → 粮税更足
                if = {
                    limit = { has_country_flag = GER_agri_tone_cooperative }
                    add_to_variable = { GER_food_surplus = 4 }
                }
                # 分地 → 粮税更少
                if = {
                    limit = { has_country_flag = GER_agri_tone_land_distributed }
                    add_to_variable = { GER_food_surplus = -2 }
                }
            }
            else = {
                # 失败：抗税/逃税，反感度上升
                add_to_variable = { GER_peasant_resentment_westphalia = 2 }
            }
            activate_mission = GER_grain_tax_mission_westphalia
        }
    }

    # =====================================================
    # 城乡交换（威斯特伐利亚）  GER_urban_rural_exchange_mission_westphalia
    #    适用：各类型地区通用（更准确的提法：城乡商品交换）
    #    效果：加粮食余量；给农业发展度；减少少量反感度
    # =====================================================
    GER_urban_rural_exchange_mission_westphalia = {
        allowed = {
            tag = GER
        }
        available = {
            always = no
        }
        activation = {
            always = no
        }
        cancel_trigger = {
            # 农业太惨 / 反感度爆表时暂停交换
            OR = {
                check_variable = { GER_peasant_resentment_westphalia > 95 }
                check_variable = { GER_agri_development < 10 }
            }
        }
        is_good = yes
        days_mission_timeout = 30
        selectable_mission = no
        timeout_effect = {
            # 基调修正：合作社 → 组织化收购，交换量更足；分地 → 农户自留多，交换量略减
            if = {
                limit = {
                    has_country_flag = GER_agri_tone_cooperative
                }
                add_to_variable = { GER_food_surplus = 9 }
                add_to_variable = { GER_agri_development = 3 }
                add_to_variable = { GER_peasant_resentment_westphalia = -3 }
            }
            else = {
                if = {
                    limit = {
                        has_country_flag = GER_agri_tone_land_distributed
                    }
                    add_to_variable = { GER_food_surplus = 4 }
                    add_to_variable = { GER_agri_development = 3 }
                    add_to_variable = { GER_peasant_resentment_westphalia = -2 }
                }
                else = {
                    add_to_variable = { GER_food_surplus = 6 }
                    add_to_variable = { GER_agri_development = 3 }
                    add_to_variable = { GER_peasant_resentment_westphalia = -2 }
                }
            }
            # 经济恢复国策解锁后：工业品充裕，交换量翻倍
            if = {
                limit = {
                    has_completed_focus = GER_economy_recovery
                }
                add_to_variable = { GER_food_surplus = 6 }
            }
            activate_mission = GER_urban_rural_exchange_mission_westphalia
        }
    }
