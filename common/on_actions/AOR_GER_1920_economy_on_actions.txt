on_actions = {
	on_daily_GER = {
		effect = {
			if = {
				limit = {
					has_focus_tree = GER_intermediate
					AOR_GER_1920_economy_visible = yes
					NOT = { has_country_flag = AOR_GER_1920_economy_layout_refreshed }
				}
				set_country_flag = AOR_GER_1920_economy_layout_refreshed
				mark_focus_tree_layout_dirty = yes
			}
			if = {
				limit = {
					has_focus_tree = GER_intermediate
					AOR_GER_post_victory_visible = yes
					NOT = { has_country_flag = AOR_GER_post_victory_layout_refreshed }
				}
				set_country_flag = AOR_GER_post_victory_layout_refreshed
				mark_focus_tree_layout_dirty = yes
			}
			# 地区精神「经济重整」：给新到手（或刚降到 5 级以下）的地区补挂；基建升满由 remove_trigger 自动摘除
			if = {
				limit = { has_country_flag = GER_1920_poor_state_active }
				every_owned_state = {
					limit = {
						infrastructure < 5
						NOT = { has_dynamic_modifier = { modifier = GER_1920_poor_state } }
					}
					add_dynamic_modifier = { modifier = GER_1920_poor_state }
				}
			}
			# 政治顾问内阁效果：条件加成、计时与每周社会化进度
			GER_adv_cabinet_daily_effects = yes
		}
	}
}
