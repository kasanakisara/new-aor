####################################################################
# 德国 · 凡尔赛条约事件链的日期驱动
# 1919.12.10：
#   1) 置旗标 AOR_GER_versailles_started（链条已启动，防重复）
#      + AOR_GER_versailles_started_flag（国策树里三个里程碑国策的 available 条件）
#   2) 当天直接触发 AOR_GER_versailles.1（葡萄酒与香烟）
#   3) 事件 immediate 显示独立国策分支，排定 12.31 的事件2
#
# 石勒苏益格线（2026-10 最终版，确定性）：
#   1920.8.1   通知国际委员会将于十天后接管            -> .6（无实际效果）
#   1920.8.11  州 839 与 1043 移交国际委员会（NSH）    -> .7
#   1920.9.1   北区公投（全部投向丹麦）                -> .8
#   1920.10.4  中部公投（全部留在德国）                -> .9
#   1921.1.5   结算：839 → 丹麦，1043 → 德国           -> .10
####################################################################
on_actions = {
	on_daily_GER = {
		effect = {
			if = {
				limit = {
					has_focus_tree = GER_intermediate
					date > 1919.12.9
					date < 1919.12.11
					NOT = { has_country_flag = AOR_GER_versailles_started }
				}
				set_country_flag = AOR_GER_versailles_started
				# 保留旧后续国策使用的启动旗标。
				set_country_flag = AOR_GER_versailles_started_flag
				country_event = { id = AOR_GER_versailles.1 }
			}

			# ------------------ 石勒苏益格线 ------------------
			if = {
				limit = { has_country_flag = AOR_GER_versailles_started }

				# 1920.8.1 通知（无实际效果）
				if = {
					limit = {
						date > 1920.7.31
						NOT = { has_country_flag = AOR_schleswig_notice }
					}
					set_country_flag = AOR_schleswig_notice
					country_event = { id = AOR_GER_versailles.6 }
				}

				# 1920.8.11 北石勒苏益格与弗伦斯堡移交国际委员会
				if = {
					limit = {
						date > 1920.8.10
						NOT = { has_country_flag = AOR_schleswig_handover }
					}
					set_country_flag = AOR_schleswig_handover
					country_event = { id = AOR_GER_versailles.7 }
				}

				# 1920.9.1 北区公投
				if = {
					limit = {
						date > 1920.8.31
						NOT = { has_country_flag = AOR_schleswig_north }
					}
					set_country_flag = AOR_schleswig_north
					country_event = { id = AOR_GER_versailles.8 }
				}

				# 1920.10.4 中区公投：中石勒苏益格（1043）与弗伦斯堡市（1125）同时揭晓
				if = {
					limit = {
						date > 1920.10.3
						NOT = { has_country_flag = AOR_schleswig_center }
					}
					set_country_flag = AOR_schleswig_center
					country_event = { id = AOR_GER_versailles.9 }
				}

				# 1921.1.5 最终结算
				if = {
					limit = {
						date > 1921.1.4
						NOT = { has_country_flag = AOR_schleswig_settled }
					}
					set_country_flag = AOR_schleswig_settled
					country_event = { id = AOR_GER_versailles.10 }
				}
			}
		}
	}
}
