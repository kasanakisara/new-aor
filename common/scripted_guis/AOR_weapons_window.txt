scripted_gui = {

	### 注入到政治界面的入口按钮
	### 注意：context_type 必须写在 window_name 之前，且取值只能是
	### player_context / decision_category / selected_country_context / country_mapicon 等；
	### 原来写的 country_context 不是合法值，会导致整个文件解析失败。
	AOR_weapons_button = {
		context_type = player_context
		window_name = "AOR_weapons_window_button"
		parent_window_name = "countrypoliticsview"

		visible = {
			always = yes
		}

		effects = {
			AOR_weapons_button_click = {
				if = {
					limit = {
						NOT = {
							has_country_flag = open_AOR_weapons_button_window
						}
					}
					set_country_flag = open_AOR_weapons_button_window
				}
				else = {
					clr_country_flag = open_AOR_weapons_button_window
				}
			}
		}
	}

	### 武器窗口本体，靠上面的国策标记控制显隐
	AOR_weapons_window = {
		context_type = player_context
		window_name = "AOR_weapons_window"

		visible = {
			has_country_flag = open_AOR_weapons_button_window
		}
	}
}
