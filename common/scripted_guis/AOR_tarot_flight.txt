    AOR_tarot_flight_gui = {
        context_type = player_context
        window_name = "AOR_tarot_flight_window"
        ai_enabled = { always = no }
        visible = { has_country_flag = AOR_tarot_flight_open }
        triggers = {
            AOR_tarot_flight_confirm_click_enabled = {
                has_country_flag = AOR_tarot_flight_played
                NOT = { has_country_flag = AOR_tarot_flight_confirmed }
                check_variable = { AOR_tarot_flight_selected > 0 }
                check_variable = { AOR_tarot_flight_selected < 9 }
            }
        }
        effects = {
            AOR_tarot_flight_confirm_click = { AOR_tarot_flight_use_selected = yes }
            AOR_tarot_flight_close_click = {
                clr_country_flag = AOR_tarot_flight_open
            }
            AOR_tarot_flight_draw_click = {
                set_variable = { AOR_tarot_flight_selected = 0 }
                AOR_tarot_flight_draw_four = yes
                if = {
                    limit = { has_country_flag = AOR_tarot_flight_variant }
                    clr_country_flag = AOR_tarot_flight_variant
                }
                else = { set_country_flag = AOR_tarot_flight_variant }
                set_country_flag = AOR_tarot_flight_played
            }
        }
    }
