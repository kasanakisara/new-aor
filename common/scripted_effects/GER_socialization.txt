# Country scope. Call after changing overview metrics for an immediate GUI update.
# Persistent frame values avoid relying on temporary data from icon visibility checks.
GER_socialization_refresh_bars = {
    GER_socialization_refresh_targets = yes
    set_variable = { GER_soc_mass_strike_cost = 350 }
    if = { limit = { has_country_flag = GER_socialization_union_reform_success } set_variable = { GER_soc_mass_strike_cost = 250 } }
    if = {
        limit = { NOT = { has_variable = GER_socialization_radicalism_base } }
        set_variable = { GER_socialization_radicalism_base = GER_socialization_radicalism }
        if = { limit = { has_variable = GER_socialization_organization_applied_radicalism_bonus } subtract_from_variable = { var = GER_socialization_radicalism_base value = GER_socialization_organization_applied_radicalism_bonus } }
        if = { limit = { has_variable = GER_socialization_radicalism_applied_bonus } subtract_from_variable = { var = GER_socialization_radicalism_base value = GER_socialization_radicalism_applied_bonus } }
        clamp_variable = { var = GER_socialization_radicalism_base min = 0 max = 100 }
    }
    # Preserve progress added directly to the displayed variable by other scripts.
    if = {
        limit = {
            has_variable = GER_socialization_radicalism_last_displayed
            OR = {
                check_variable = { GER_socialization_radicalism > GER_socialization_radicalism_last_displayed }
                check_variable = { GER_socialization_radicalism < GER_socialization_radicalism_last_displayed }
            }
        }
        set_variable = { GER_socialization_radicalism_external_change = GER_socialization_radicalism }
        subtract_from_variable = { var = GER_socialization_radicalism_external_change value = GER_socialization_radicalism_last_displayed }
        add_to_variable = { var = GER_socialization_radicalism_base value = GER_socialization_radicalism_external_change }
    }
    if = { limit = { NOT = { has_variable = GER_socialization_organization_level } } set_variable = { GER_socialization_organization_level = 0 } }
    clamp_variable = { var = GER_socialization_organization_level min = 0 max = 4 }
    GER_socialization_organization_borrow_level = yes
    GER_socialization_organization_borrow_level = yes
    GER_socialization_organization_borrow_level = yes
    GER_socialization_organization_borrow_level = yes
    if = { limit = { check_variable = { GER_socialization_organization < 0 } } set_variable = { GER_socialization_organization = 0 } }
    # The bar tracks progress within the current level. Carry any overflow forward.
    GER_socialization_organization_advance_level = yes
    GER_socialization_organization_advance_level = yes
    GER_socialization_organization_advance_level = yes
    GER_socialization_organization_advance_level = yes
    if = { limit = { check_variable = { GER_socialization_organization_level = 4 } } clamp_variable = { var = GER_socialization_organization min = 0 max = 100 } }
    GER_socialization_refresh_organization_effect = yes
    GER_socialization_radicalism_borrow_level = yes
    GER_socialization_radicalism_borrow_level = yes
    GER_socialization_radicalism_borrow_level = yes
    if = { limit = { check_variable = { GER_socialization_radicalism_base < 0 } } set_variable = { GER_socialization_radicalism_base = 0 } }
    GER_socialization_refresh_radicalism_effect = yes
    set_variable = { GER_socialization_radicalism = GER_socialization_radicalism_base }
    add_to_variable = { var = GER_socialization_radicalism value = GER_socialization_organization_new_radicalism_bonus }
    add_to_variable = { var = GER_socialization_radicalism value = GER_socialization_radicalism_new_bonus }
    GER_socialization_radicalism_advance_level = yes
    GER_socialization_radicalism_advance_level = yes
    GER_socialization_radicalism_advance_level = yes
    clamp_variable = { var = GER_socialization_radicalism min = 0 max = 100 }
    set_variable = { GER_socialization_radicalism_last_displayed = GER_socialization_radicalism }
    set_variable = { GER_socialization_organization_fill = GER_socialization_organization }
    clamp_variable = { var = GER_socialization_organization_fill min = 0 max = 100 }
    clamp_variable = { var = GER_socialization_counterrevolution min = 0 max = 1200 }
    set_variable = { GER_socialization_counterrevolution_fill = GER_socialization_counterrevolution }
    divide_variable = { GER_socialization_counterrevolution_fill = 12 }
    clamp_variable = { var = GER_socialization_counterrevolution_fill min = 0 max = 100 }
    set_variable = { GER_socialization_counterrevolution_percent = GER_socialization_counterrevolution_fill }
    # The automatic-action script can gate its choices on this intensity variable.
    # 0: none; 1: <25%; 2: 25-74%; 3: 75-99%; 4: 100% (all actions).
    set_variable = { GER_socialization_counterrevolution_intensity = 0 }
    if = { limit = { check_variable = { GER_socialization_counterrevolution_percent > 0 } } set_variable = { GER_socialization_counterrevolution_intensity = 1 } }
    if = { limit = { check_variable = { GER_socialization_counterrevolution > 299 } } set_variable = { GER_socialization_counterrevolution_intensity = 2 } }
    if = { limit = { check_variable = { GER_socialization_counterrevolution > 899 } } set_variable = { GER_socialization_counterrevolution_intensity = 3 } }
    if = { limit = { check_variable = { GER_socialization_counterrevolution > 1199 } } set_variable = { GER_socialization_counterrevolution_intensity = 4 } }
    set_variable = { GER_socialization_radicalism_fill = GER_socialization_radicalism }
    clamp_variable = { var = GER_socialization_radicalism_fill min = 0 max = 100 }
    GER_socialization_distribute_points = yes
    GER_socialization_refresh_selected_industry = yes
}

# One entry point for the schedule. A future focus can call this instead of on_startup.
GER_socialization_initialize = {
    if = {
        limit = { NOT = { has_country_flag = GER_socialization_started } }
        set_country_flag = GER_socialization_started
        # Starting balance; do not overwrite an existing save's socialization values.
        if = { limit = { NOT = { has_variable = GER_socialization_counterrevolution } } set_variable = { GER_socialization_counterrevolution = 1000 } }
        if = { limit = { NOT = { has_variable = GER_socialization_organization_level } } set_variable = { GER_socialization_organization_level = 0 } }
        if = { limit = { NOT = { has_variable = GER_socialization_organization } } set_variable = { GER_socialization_organization = 50 } }
        if = { limit = { NOT = { has_variable = GER_socialization_radicalism_level } } set_variable = { GER_socialization_radicalism_level = 0 } }
        if = { limit = { NOT = { has_variable = GER_socialization_radicalism_base } } set_variable = { GER_socialization_radicalism_base = 50 } }
        if = { limit = { NOT = { has_variable = GER_socialization_radicalism } } set_variable = { GER_socialization_radicalism = 50 } }
        if = { limit = { NOT = { has_variable = GER_socialization_selected } } set_variable = { GER_socialization_selected = 0 } }
        if = { limit = { NOT = { has_variable = GER_socialization_active_industry } } set_variable = { GER_socialization_active_industry = 0 } }
        if = { limit = { NOT = { has_variable = GER_socialization_days_remaining } } set_variable = { GER_socialization_days_remaining = 65 } }
        if = { limit = { NOT = { has_variable = GER_socialization_elapsed_days } } set_variable = { GER_socialization_elapsed_days = 0 } }
        GER_soc_auto_reset_industry = yes
    }
    GER_socialization_refresh_bars = yes
}

# One day passes for the current industry. A finished industry advances during
# refresh; an unfinished industry fails when its remaining days reach zero.
GER_socialization_daily_tick = {
    if = { limit = { NOT = { has_country_flag = GER_soc_auto_initialized } } GER_soc_auto_reset_industry = yes set_country_flag = GER_soc_auto_initialized }
    GER_socialization_refresh_bars = yes
    GER_socialization_reconcile_rewards = yes
    if = {
        limit = { has_country_flag = GER_socialization_reward_mining }
        add_to_variable = { GER_socialization_mining_refresh_days = 1 }
        if = {
            limit = { check_variable = { GER_socialization_mining_refresh_days > 29 } }
            subtract_from_variable = { GER_socialization_mining_refresh_days = 30 }
            GER_socialization_refresh_mining_resources = yes
        }
    }
    if = {
        limit = { has_country_flag = GER_socialization_reward_banking check_variable = { GER_socialization_active_industry < 12 } }
        add_to_variable = { GER_socialization_bank_bonus_days = 1 }
        if = {
            limit = { check_variable = { GER_socialization_bank_bonus_days > 6 } }
            subtract_from_variable = { GER_socialization_bank_bonus_days = 7 }
            add_to_variable = { GER_socialization_points_pool = 4 }
            GER_socialization_refresh_bars = yes
        }
    }
    set_variable = { GER_socialization_day_start_industry = GER_socialization_active_industry }
    if = { limit = { check_variable = { GER_socialization_day_start_industry < 12 } } add_to_variable = { GER_socialization_elapsed_days = 1 } }
    GER_socialization_decisions_daily_tick = yes
    GER_soc_auto_daily_tick = yes
    if = {
        limit = {
            check_variable = { GER_socialization_active_industry < 12 }
            check_variable = { GER_socialization_active_industry = GER_socialization_day_start_industry }
            NOT = { check_variable = { GER_socialization_active_industry > GER_socialization_day_start_industry } }
        }
        subtract_from_variable = { GER_socialization_days_remaining = 1 }
        if = {
            limit = { check_variable = { GER_socialization_days_remaining < 1 } }
            GER_socialization_advance_industry = yes
        }
    }
    GER_socialization_refresh_selected_industry = yes
}
