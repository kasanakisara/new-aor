# The following scripted effect does the main job of generating the gui, by adding the seats for each party in a grid.
# This should be called every time you need to update the gui, when you add/subtract seats for example.
# Read the comments if you're interested in how it works
TUR_reload_parliament_gui = {
	# Clears what's currently on the gui, if there's anything there.
	clear_array = TUR_parliament_main_array

	# The array temp_seat is duplicated from the array TUR_parliament_seat_array.
	for_each_loop = {
		array = TUR_parliament_seat_array

		add_to_array = { temp_seat = TUR_parliament_seat_array^i }
	}

	# It's a suprise tool for later
	set_temp_variable = { temp_index_1 = 0 }
	set_temp_variable = { temp_index_2 = 1 }

	# Set temp_size as the highest index in the array, highest index is the highest value x in temp_array^x.
	set_temp_variable = { temp_size = temp_seat^num }
	subtract_from_temp_variable = { temp_size = 1 }

	# What this does is loop through the temp_seat array, adding the current value to the next value on the list.
	# temp_seat^0 is the first value on the list, temp_seat^1 is the next and so forth.
	# the loop ends when temp_index_1 is equal to the size of the array - 1 (basically the max index).
	while_loop_effect = {
		limit = {
			check_variable = { temp_index_1 < temp_size }
		}
		# The suprise tool starts out as temp_seat^0 and temp_seat^1 and then is incremented every loop , adding the current value to the next value on the list.
		add_to_variable = { temp_seat^temp_index_2 = temp_seat^temp_index_1 }
		add_to_temp_variable = { temp_index_1 = 1 }
		add_to_temp_variable = { temp_index_2 = 1 }
	}

	# resize_array fills the TUR_parliament_member_array with the id of the parliament member x amount of times where x is the number of seats the member has.
	# so the array will look like this basically:
	# 1,1,1,1,1,1,1,...(repeated 90 times),2,2,2,2,2,2,2(repeated 35 times),3,3,3.. etc.
	for_loop_effect = {
		start = 0
		end = temp_seat^num

		resize_array = { array = TUR_parliament_main_array value = TUR_parliament_member_array^v size = temp_seat^v }
	}

	# Clear the temp_seat array used previously since it's not needed anymore.
	clear_array = temp_seat
}

TUR_reelect_parliament_seats = {
	TUR = {
		# --- 第一步：计算总剩余值 ---
		set_variable = { var = TUR_election_total_remaining value = TUR_seatNumbers }
		subtract_from_variable = { var = TUR_election_total_remaining value = TUR_Vacant }
		subtract_from_variable = { var = TUR_election_total_remaining value = TUR_Ahrar }
		subtract_from_variable = { var = TUR_election_total_remaining value = TUR_Hukuk }
		subtract_from_variable = { var = TUR_election_total_remaining value = TUR_Nonpartisan }

		# --- 第二步：计算三组权重 ---
		# 权重1
		set_variable = { var = TUR_election_weight_1 value = party_popularity@AUT_democratic }
		add_to_variable = { var = TUR_election_weight_1 value = party_popularity@SOC_democratic }
		add_to_variable = { var = TUR_election_weight_1 value = party_popularity@Absolutism }

		# 权重2
		set_variable = { var = TUR_election_weight_2 value = party_popularity@FRE_democratic }
		add_to_variable = { var = TUR_election_weight_2 value = party_popularity@CON_democratic }
		add_to_variable = { var = TUR_election_weight_2 value = party_popularity@ISL_democratic }

		# 权重3
		set_variable = { var = TUR_election_weight_3 value = party_popularity@Marxism }
		add_to_variable = { var = TUR_election_weight_3 value = party_popularity@PAR_socialism }

		# --- 第三步：计算总权重 ---
		set_variable = { var = TUR_election_total_weight value = TUR_election_weight_1 }
		add_to_variable = { var = TUR_election_total_weight value = TUR_election_weight_2 }
		add_to_variable = { var = TUR_election_total_weight value = TUR_election_weight_3 }

		# --- 第四步：按比例分配 ---
		if = {
			limit = { check_variable = { TUR_election_total_weight > 0 } }

			# TUR_election_part_1 = floor(TUR_election_weight_1 / TUR_election_total_weight × TUR_election_total_remaining)
			set_variable = { var = TUR_election_part_1 value = TUR_election_weight_1 }
			multiply_variable = { var = TUR_election_part_1 value = TUR_election_total_remaining }
			divide_variable = { var = TUR_election_part_1 value = TUR_election_total_weight }
			round_variable = { var = TUR_election_part_1 }

			# TUR_election_part_2 = floor(TUR_election_weight_2 / TUR_election_total_weight × TUR_election_total_remaining)
			set_variable = { var = TUR_election_part_2 value = TUR_election_weight_2 }
			multiply_variable = { var = TUR_election_part_2 value = TUR_election_total_remaining }
			divide_variable = { var = TUR_election_part_2 value = TUR_election_total_weight }
			round_variable = { var = TUR_election_part_2 }

			# TUR_election_part_3 = TUR_election_total_remaining - TUR_election_part_1 - TUR_election_part_2
			# 余量全部归入第三份，保证三份之和精确等于 TUR_election_total_remaining
			set_variable = { var = TUR_election_part_3 value = TUR_election_total_remaining }
			subtract_from_variable = { var = TUR_election_part_3 value = TUR_election_part_1 }
			subtract_from_variable = { var = TUR_election_part_3 value = TUR_election_part_2 }
		}
		else = {
			# 兜底处理：总权重为 0 时三等分，余量归 TUR_election_part_3
			set_variable = { var = TUR_election_part_1 value = TUR_election_total_remaining }
			divide_variable = { var = TUR_election_part_1 value = 3 }
			round_variable = { var = TUR_election_part_1 }
			set_variable = { var = TUR_election_part_2 value = TUR_election_part_1 }
			round_variable = { var = TUR_election_part_2 }
			set_variable = { var = TUR_election_part_3 value = TUR_election_total_remaining }
			subtract_from_variable = { var = TUR_election_part_3 value = TUR_election_part_1 }
			subtract_from_variable = { var = TUR_election_part_3 value = TUR_election_part_2 }
		}

		# --- 结果已存入 var:TUR_election_part_1, var:TUR_election_part_2, var:TUR_election_part_3 ---
		set_variable = { var = TUR_ITC value = TUR_election_part_1 }
		set_variable = { var = TUR_HIF value = TUR_election_part_2 }
		set_variable = { var = TUR_OSF value = TUR_election_part_3 }
	}
	clear_array = TUR_parliament_seat_array
	add_to_array = { TUR_parliament_seat_array = TUR_ITC }
	add_to_array = { TUR_parliament_seat_array = TUR_HIF }
	add_to_array = { TUR_parliament_seat_array = TUR_OSF }
	add_to_array = { TUR_parliament_seat_array = TUR_Ahrar }
	add_to_array = { TUR_parliament_seat_array = TUR_Hukuk }
	add_to_array = { TUR_parliament_seat_array = TUR_Nonpartisan }
	add_to_array = { TUR_parliament_seat_array = TUR_Vacant }
	TUR_reload_parliament_gui = yes
	add_to_variable = { var = global.AOR_TUR_parliament_gui_dirty value = 0.01 }
