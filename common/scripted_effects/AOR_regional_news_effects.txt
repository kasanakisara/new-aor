# New preferences start with every continent enabled. Existing saved choices stay unchanged.
AOR_news_initialize_preferences = {
    if = {
        limit = { NOT = { has_country_flag = AOR_news_preferences_initialized } }
        set_country_flag = AOR_news_subscribed_europe
        set_country_flag = AOR_news_subscribed_asia
        set_country_flag = AOR_news_subscribed_africa
        set_country_flag = AOR_news_subscribed_oceania
        set_country_flag = AOR_news_subscribed_north_america
        set_country_flag = AOR_news_subscribed_south_america
        set_country_flag = AOR_news_preferences_initialized
    }
}
