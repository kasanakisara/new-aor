# AOR: dependency-audited scripted functions.

GER_SPD_default_support_effect = {
	set_variable = { GER_SPD_left_support = 17 }
	set_variable = { GER_SPD_center_support = 43 }
	set_variable = { GER_SPD_right_support = 40 }
}

GER_SPD_initialize_support_effect = {
	if = {
		limit = {
			NOT = { has_country_flag = GER_SPD_support_initialized }
			NOT = {
				OR = {
					has_variable = GER_SPD_left_support
					has_variable = GER_SPD_center_support
					has_variable = GER_SPD_right_support
				}
			}
		}
		GER_SPD_default_support_effect = yes
	}
	set_country_flag = GER_SPD_support_initialized
	GER_SPD_refresh_support_effect = yes
}

GER_SPD_refresh_support_effect = {
	if = {
		limit = { check_variable = { GER_SPD_left_support < 0 } }
		set_variable = { GER_SPD_left_support = 0 }
	}
	if = {
		limit = { check_variable = { GER_SPD_center_support < 0 } }
		set_variable = { GER_SPD_center_support = 0 }
	}
	if = {
		limit = { check_variable = { GER_SPD_right_support < 0 } }
		set_variable = { GER_SPD_right_support = 0 }
	}
	set_temp_variable = { GER_SPD_total = GER_SPD_left_support }
	add_to_temp_variable = { GER_SPD_total = GER_SPD_center_support }
	add_to_temp_variable = { GER_SPD_total = GER_SPD_right_support }
	if = {
		limit = { check_variable = { GER_SPD_total = 0 } }
		GER_SPD_default_support_effect = yes
		set_temp_variable = { GER_SPD_total = 100 }
	}

	# Preserve fractional percentage points (ADGB adds 0.3 every fortnight).
	set_temp_variable = { GER_SPD_normalization_factor = 100 }
	divide_temp_variable = { GER_SPD_normalization_factor = GER_SPD_total }
	multiply_variable = { GER_SPD_left_support = GER_SPD_normalization_factor }
	multiply_variable = { GER_SPD_center_support = GER_SPD_normalization_factor }
	set_variable = { GER_SPD_right_support = 100 }
	subtract_from_variable = { GER_SPD_right_support = GER_SPD_left_support }
	subtract_from_variable = { GER_SPD_right_support = GER_SPD_center_support }
	set_variable = { GER_SPD_left_last_support = GER_SPD_left_support }
	GER_SPD_refresh_display_support_effect = yes
	set_variable = { GER_SPD_center_last_support = GER_SPD_center_support }
	set_variable = { GER_SPD_right_last_support = GER_SPD_right_support }
	add_to_variable = { GER_SPD_gui_revision = 1 }
}
