# 把 yh_add_id 放进议程表，状态 yh_add_st（0 未解锁 / 1 可处理）
GER_yihui_put_agenda = {
	if = {
		limit = { check_variable = { yh_add_id > 0 } }
		GER_yihui_ensure_size = yes
		if = {
			limit = { check_variable = { GER_yihui_ag_state^yh_add_id = -1 } }
			set_variable = { GER_yihui_ag_state^yh_add_id = yh_add_st }
			set_variable = { GER_yihui_ag_days^yh_add_id = 0 }
			set_variable = { GER_yihui_ag_total^yh_add_id = 0 }
			set_variable = { GER_yihui_agenda_prog^yh_add_id = 0 }
			add_to_array = { array = GER_yihui_agenda_list value = yh_add_id }
		}
		else_if = {
			limit = {
				check_variable = { GER_yihui_ag_state^yh_add_id = 0 }
				check_variable = { yh_add_st = 1 }
			}
			set_variable = { GER_yihui_ag_state^yh_add_id = 1 }
		}
		GER_yihui_sync = yes
	}
}

# 按当前勾选算预计耗时和收益
GER_yihui_recalc_preview = {
	set_temp_variable = { yh_n = 0 }
	add_to_temp_variable = { yh_n = GER_yihui_pick_1 }
	add_to_temp_variable = { yh_n = GER_yihui_pick_2 }
	add_to_temp_variable = { yh_n = GER_yihui_pick_3 }
	add_to_temp_variable = { yh_n = GER_yihui_pick_4 }
	add_to_temp_variable = { yh_n = GER_yihui_pick_5 }
	add_to_temp_variable = { yh_n = GER_yihui_pick_6 }
	add_to_temp_variable = { yh_n = GER_yihui_pick_7 }
	add_to_temp_variable = { yh_n = GER_yihui_pick_8 }
	set_temp_variable = { yh_g = 0 }
	if = { limit = { OR = { check_variable = { GER_yihui_pick_1 = 1 } check_variable = { GER_yihui_pick_2 = 1 } check_variable = { GER_yihui_pick_3 = 1 } } } add_to_temp_variable = { yh_g = 1 } }  # spd
	if = { limit = { OR = { check_variable = { GER_yihui_pick_4 = 1 } } } add_to_temp_variable = { yh_g = 1 } }  # z
	if = { limit = { OR = { check_variable = { GER_yihui_pick_5 = 1 } } } add_to_temp_variable = { yh_g = 1 } }  # ddp
	if = { limit = { OR = { check_variable = { GER_yihui_pick_6 = 1 } } } add_to_temp_variable = { yh_g = 1 } }  # dvp
	if = { limit = { OR = { check_variable = { GER_yihui_pick_8 = 1 } } } add_to_temp_variable = { yh_g = 1 } }  # other
	set_temp_variable = { yh_c = 0 }  # 执政联盟党团数
	add_to_temp_variable = { yh_c = GER_yihui_pick_1 }
	add_to_temp_variable = { yh_c = GER_yihui_pick_2 }
	add_to_temp_variable = { yh_c = GER_yihui_pick_3 }
	add_to_temp_variable = { yh_c = GER_yihui_pick_4 }
	set_temp_variable = { yh_o = yh_n }  # 反对党团数
	subtract_from_temp_variable = { yh_o = yh_c }
	set_variable = { GER_yihui_preview_n = yh_n }
	if = {
		limit = { check_variable = { yh_n > 0 } }
		# 天数 = 25 + 同党多一派 5 天 + 跨党多一党 10 天
		set_temp_variable = { yh_same = yh_n }
		subtract_from_temp_variable = { yh_same = yh_g }
		multiply_temp_variable = { yh_same = 5 }
		set_temp_variable = { yh_cross = yh_g }
		subtract_from_temp_variable = { yh_cross = 1 }
		multiply_temp_variable = { yh_cross = 10 }
		set_variable = { GER_yihui_preview_days = 25 }
		add_to_variable = { GER_yihui_preview_days = yh_same }
		add_to_variable = { GER_yihui_preview_days = yh_cross }
		# 协商惯性 +（派数 - 1）+ 反对党团数 × 1：拉的人越多、越跨阵营，议会越习惯协商
		set_variable = { GER_yihui_preview_inertia = yh_n }
		subtract_from_variable = { GER_yihui_preview_inertia = 1 }
		set_temp_variable = { yh_x = yh_o }
		multiply_temp_variable = { yh_x = 1 }
		add_to_variable = { GER_yihui_preview_inertia = yh_x }
		# 联盟凝聚力 + 执政联盟党团数 - 反对党团数 × 2：把事交给反对党，联盟伙伴会不满
		set_variable = { GER_yihui_preview_cohesion = yh_c }
		set_temp_variable = { yh_x = yh_o }
		multiply_temp_variable = { yh_x = 2 }
		subtract_from_variable = { GER_yihui_preview_cohesion = yh_x }
		# 雨果·普罗伊斯：委派德意志民主党（5）时免除联盟凝聚力惩罚
		if = {
			limit = {
				check_variable = { GER_yihui_pick_5 = 1 }
				GER_03_08_Hugo_Preuss = { is_hired_as_advisor = yes }
			}
			add_to_variable = { GER_yihui_preview_cohesion = 2 }
		}
		# 古斯塔夫·施特雷泽曼：委派德意志人民党（6）时免除联盟凝聚力惩罚
		if = {
			limit = {
				check_variable = { GER_yihui_pick_6 = 1 }
				GER_12_01_Gustav_Stresemann = { is_hired_as_advisor = yes }
			}
			add_to_variable = { GER_yihui_preview_cohesion = 2 }
		}
	}
	else = {
		set_variable = { GER_yihui_preview_days = 0 }
		set_variable = { GER_yihui_preview_inertia = 0 }
		set_variable = { GER_yihui_preview_cohesion = 0 }
	}
}

# 刷新界面用的显示变量（打开界面、点击、每日计时后都会调用）
GER_yihui_sync = {
	set_variable = { GER_yihui_sel_state = -1 }
	set_temp_variable = { yh_sid = GER_yihui_selected_agenda }
	if = {
		limit = {
			check_variable = { yh_sid > 0 }
			check_variable = { GER_yihui_ag_state^num > yh_sid }
		}
		set_variable = { GER_yihui_sel_state = GER_yihui_ag_state^yh_sid }
	}
	# 只有可处理 / 处理中的议程能保持选中；已通过、被撤下的取消选中
	if = {
		limit = { NOT = { OR = { check_variable = { GER_yihui_sel_state = 1 } check_variable = { GER_yihui_sel_state = 2 } } } }
		if = {
			limit = { NOT = { check_variable = { GER_yihui_selected_agenda = 0 } } }
			set_variable = { GER_yihui_selected_agenda = 0 }
			GER_yihui_clear_picks = yes
		}
		set_variable = { GER_yihui_sel_state = -1 }
	}
	clear_array = GER_yihui_view_picks
	GER_yihui_recalc_preview = yes
	if = {
		limit = { check_variable = { GER_yihui_sel_state = 1 } }
		if = { limit = { check_variable = { GER_yihui_pick_1 = 1 } } add_to_array = { GER_yihui_view_picks = 1 } }
		if = { limit = { check_variable = { GER_yihui_pick_2 = 1 } } add_to_array = { GER_yihui_view_picks = 2 } }
		if = { limit = { check_variable = { GER_yihui_pick_3 = 1 } } add_to_array = { GER_yihui_view_picks = 3 } }
		if = { limit = { check_variable = { GER_yihui_pick_4 = 1 } } add_to_array = { GER_yihui_view_picks = 4 } }
		if = { limit = { check_variable = { GER_yihui_pick_5 = 1 } } add_to_array = { GER_yihui_view_picks = 5 } }
		if = { limit = { check_variable = { GER_yihui_pick_6 = 1 } } add_to_array = { GER_yihui_view_picks = 6 } }
		if = { limit = { check_variable = { GER_yihui_pick_7 = 1 } } add_to_array = { GER_yihui_view_picks = 7 } }
		if = { limit = { check_variable = { GER_yihui_pick_8 = 1 } } add_to_array = { GER_yihui_view_picks = 8 } }
		set_variable = { GER_yihui_sel_days = GER_yihui_preview_days }
		set_variable = { GER_yihui_sel_inertia = GER_yihui_preview_inertia }
		set_variable = { GER_yihui_sel_cohesion = GER_yihui_preview_cohesion }

		set_variable = { GER_yihui_sel_cohesion = GER_yihui_ag_gain_c^yh_sid }
		set_variable = { GER_yihui_sel_prog = GER_yihui_agenda_prog^yh_sid }
		if = { limit = { check_variable = { GER_yihui_ag_p1^yh_sid = 1 } } add_to_array = { GER_yihui_view_picks = 1 } }
		if = { limit = { check_variable = { GER_yihui_ag_p2^yh_sid = 1 } } add_to_array = { GER_yihui_view_picks = 2 } }
		if = { limit = { check_variable = { GER_yihui_ag_p3^yh_sid = 1 } } add_to_array = { GER_yihui_view_picks = 3 } }

		set_variable = { GER_yihui_ag_days^yh_id = GER_yihui_preview_days }
		set_variable = { GER_yihui_ag_total^yh_id = GER_yihui_preview_days }

# 重算处理中的议程数和各党团是否在忙
GER_yihui_recount_busy = {
	set_variable = { GER_yihui_running_count = 0 }
	set_variable = { GER_yihui_busy_1 = 0 }
	set_variable = { GER_yihui_busy_2 = 0 }
	set_variable = { GER_yihui_busy_3 = 0 }
	set_variable = { GER_yihui_busy_4 = 0 }
	set_variable = { GER_yihui_busy_5 = 0 }
	set_variable = { GER_yihui_busy_6 = 0 }
	set_variable = { GER_yihui_busy_7 = 0 }
	set_variable = { GER_yihui_busy_8 = 0 }
	for_each_loop = {
		array = GER_yihui_agenda_list
		value = yh_id
		index = yh_idx
		if = {
			limit = { check_variable = { GER_yihui_ag_state^yh_id = 2 } }
			add_to_variable = { GER_yihui_running_count = 1 }
			if = { limit = { check_variable = { GER_yihui_ag_p1^yh_id = 1 } } set_variable = { GER_yihui_busy_1 = 1 } }
			if = { limit = { check_variable = { GER_yihui_ag_p2^yh_id = 1 } } set_variable = { GER_yihui_busy_2 = 1 } }
			if = { limit = { check_variable = { GER_yihui_ag_p3^yh_id = 1 } } set_variable = { GER_yihui_busy_3 = 1 } }
			if = { limit = { check_variable = { GER_yihui_ag_p4^yh_id = 1 } } set_variable = { GER_yihui_busy_4 = 1 } }
			if = { limit = { check_variable = { GER_yihui_ag_p5^yh_id = 1 } } set_variable = { GER_yihui_busy_5 = 1 } }
			if = { limit = { check_variable = { GER_yihui_ag_p6^yh_id = 1 } } set_variable = { GER_yihui_busy_6 = 1 } }
			if = { limit = { check_variable = { GER_yihui_ag_p7^yh_id = 1 } } set_variable = { GER_yihui_busy_7 = 1 } }
			if = { limit = { check_variable = { GER_yihui_ag_p8^yh_id = 1 } } set_variable = { GER_yihui_busy_8 = 1 } }
		}
	}
	# 党团在处理议程期间提供的加成（同一党派同时处理几项也只算一次）
	set_variable = { GER_yihui_bonus_factory = 0 }
	if = { limit = { check_variable = { GER_yihui_busy_1 > 0 } } set_variable = { GER_yihui_bonus_factory = 0.2 } }
	set_variable = { GER_yihui_bonus_stability = 0 }
	if = { limit = { check_variable = { GER_yihui_busy_3 > 0 } } set_variable = { GER_yihui_bonus_stability = 0.001 } }
	set_variable = { GER_yihui_bonus_pp = 0 }
	if = { limit = { check_variable = { GER_yihui_busy_4 > 0 } } set_variable = { GER_yihui_bonus_pp = 0.1 } }
	set_variable = { GER_yihui_bonus_efficiency = 0 }
	if = { limit = { check_variable = { GER_yihui_busy_5 > 0 } } set_variable = { GER_yihui_bonus_efficiency = 0.15 } }
	set_variable = { GER_yihui_bonus_building = 0 }
	if = { limit = { check_variable = { GER_yihui_busy_6 > 0 } } set_variable = { GER_yihui_bonus_building = 0.15 } }
	set_variable = { GER_yihui_bonus_local = 0 }
	if = { limit = { check_variable = { GER_yihui_busy_8 > 0 } } set_variable = { GER_yihui_bonus_local = 0.03 } }
	# 党团协作只在有议程处理时挂着
	if = {
		limit = { check_variable = { GER_yihui_running_count > 0 } }
		if = { limit = { NOT = { has_dynamic_modifier = { modifier = GER_yihui_party_cooperation } } } add_dynamic_modifier = { modifier = GER_yihui_party_cooperation } }
		force_update_dynamic_modifier = yes
	}
	else_if = {
		limit = { has_dynamic_modifier = { modifier = GER_yihui_party_cooperation } }
		remove_dynamic_modifier = { modifier = GER_yihui_party_cooperation }
	}
	# YIHUI_DEBUG 临时排查，确认后删除
	if = { limit = { has_dynamic_modifier = { modifier = GER_yihui_party_cooperation } } log = "YIHUI_DEBUG [GetDateText] recount running=[?GER_yihui_running_count] coop=ON factory=[?GER_yihui_bonus_factory] stab=[?GER_yihui_bonus_stability]" }
	else = { log = "YIHUI_DEBUG [GetDateText] recount running=[?GER_yihui_running_count] coop=OFF" }
}

# 每日计时：处理中的议程天数 -1，到期的先记进 GER_yihui_done_queue，循环结束后再结算（结算会改议程表）
GER_yihui_daily_tick = {
	if = {
		limit = {
			has_country_flag = GER_yihui_agenda_v2
			check_variable = { GER_yihui_running_count > 0 }
		}
		clear_array = GER_yihui_done_queue
		# 有议程在处理却没挂党团协作（热重载 / 旧存档）时补挂
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = GER_yihui_party_cooperation } } }
			log = "YIHUI_DEBUG [GetDateText] daily: coop missing, recount"
			GER_yihui_recount_busy = yes
		}
		for_each_loop = {
			array = GER_yihui_agenda_list
			value = yh_id
			index = yh_idx
			if = {
				limit = { check_variable = { GER_yihui_ag_state^yh_id = 2 } }
				set_temp_variable = { yh_d = GER_yihui_ag_days^yh_id }
				subtract_from_temp_variable = { yh_d = 1 }
				set_variable = { GER_yihui_ag_days^yh_id = yh_d }
				set_temp_variable = { yh_t = GER_yihui_ag_total^yh_id }
				if = {
					limit = { check_variable = { yh_t > 0 } }
					set_temp_variable = { yh_p = yh_t }
					subtract_from_temp_variable = { yh_p = yh_d }
					multiply_temp_variable = { yh_p = 100 }
					divide_temp_variable = { yh_p = yh_t }
					set_variable = { GER_yihui_agenda_prog^yh_id = yh_p }
				}
				if = {
					limit = { check_variable = { yh_d < 1 } }
					add_to_array = { array = GER_yihui_done_queue value = yh_id }
				}
			}
		}
		if = {
			limit = { check_variable = { GER_yihui_done_queue^num > 0 } }
			for_each_loop = {
				array = GER_yihui_done_queue
				value = yh_id
				index = yh_idx
				GER_yihui_complete_agenda = yes
			}
			clear_array = GER_yihui_done_queue
			GER_yihui_recount_busy = yes
		}
		GER_yihui_sync = yes
	}
}

# 结算议程 yh_id（每日计时里调用）
GER_yihui_complete_agenda = {
	set_variable = { GER_yihui_ag_state^yh_id = 3 }
	set_variable = { GER_yihui_ag_days^yh_id = 0 }
	set_variable = { GER_yihui_agenda_prog^yh_id = 100 }
	set_temp_variable = { yh_x = GER_yihui_ag_gain_i^yh_id }
	add_to_variable = { GER_yihui_inertia = yh_x }
	clamp_variable = { var = GER_yihui_inertia min = 0 max = 100 }
	set_temp_variable = { yh_x = GER_yihui_ag_gain_c^yh_id }
	add_to_variable = { GER_yihui_cohesion = yh_x }
	clamp_variable = { var = GER_yihui_cohesion min = 0 max = 100 }
	if = { limit = { GER_10_02_Matthias_Erzberger = { is_hired_as_advisor = yes } } add_political_power = 15 }
	if = { limit = { check_variable = { GER_yihui_ag_p2^yh_id = 1 } } add_political_power = 50 }  # SPD 中派：政治点数 +50
	set_variable = { GER_yihui_ag_p1^yh_id = 0 }
	set_variable = { GER_yihui_ag_p2^yh_id = 0 }
	set_variable = { GER_yihui_ag_p3^yh_id = 0 }
	set_variable = { GER_yihui_ag_p4^yh_id = 0 }
	set_variable = { GER_yihui_ag_p5^yh_id = 0 }
	set_variable = { GER_yihui_ag_p6^yh_id = 0 }
	set_variable = { GER_yihui_ag_p7^yh_id = 0 }
	set_variable = { GER_yihui_ag_p8^yh_id = 0 }
	remove_from_array = { array = GER_yihui_agenda_list value = yh_id }
	if = {
		limit = { NOT = { is_in_array = { GER_yihui_passed_list = yh_id } } }
		add_to_array = { array = GER_yihui_passed_list value = yh_id }
	}
	# 各议程自己的完成效果在 GER_yihui_agenda_results.txt：GER_yihui_agenda_<编号>_result
	set_variable = { GER_yihui_last_passed = yh_id }
	meta_effect = {
		text = { GER_yihui_agenda_[ID]_result = yes }
		ID = "[?GER_yihui_last_passed]"
	}
}
