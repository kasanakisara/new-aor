GER_adv_mehring_term = {
	if = { limit = { NOT = { has_variable = GER_adv_mehring_days } } set_variable = { var = GER_adv_mehring_days value = 0 } }
	if = {
		limit = { GER_01_02_Franz_Mehring = { is_hired_as_advisor = yes } }
		if = {
			limit = { NOT = { has_country_flag = GER_adv_mehring_hired } }
			set_country_flag = GER_adv_mehring_hired
			set_variable = { var = GER_adv_mehring_days value = 0 }
		}
		add_to_variable = { var = GER_adv_mehring_days value = 1 }
		# 100 天：政治点数 +5%
		if = {
			limit = {
				check_variable = { var = GER_adv_mehring_days value = 100 compare = more_than_or_equals }
				NOT = { has_country_flag = GER_adv_mehring_pp_unlocked }
			}
			set_country_flag = GER_adv_mehring_pp_unlocked
			add_dynamic_modifier = { modifier = GER_adv_mehring_pp }
		}
		# 120 天：社会化 +2/周（里程碑，不自动解职；解职后重新聘用，天数从 0 重新计算）
		if = {
			limit = {
				check_variable = { var = GER_adv_mehring_days value = 120 compare = more_than_or_equals }
				NOT = { has_country_flag = GER_adv_mehring_soc_unlocked }
			}
			set_country_flag = GER_adv_mehring_soc_unlocked
		}
	}
	else = {
		if = {
			limit = { has_country_flag = GER_adv_mehring_hired }
			clr_country_flag = GER_adv_mehring_hired
			set_variable = { var = GER_adv_mehring_days value = 0 }
		}
		# 解职后里程碑效果失效；重新聘用从 0 天重新计时
		if = {
			limit = { has_country_flag = GER_adv_mehring_pp_unlocked }
			clr_country_flag = GER_adv_mehring_pp_unlocked
			remove_dynamic_modifier = { modifier = GER_adv_mehring_pp }
		}
		if = {
			limit = { has_country_flag = GER_adv_mehring_soc_unlocked }
