# 界面刷新（scripted_gui 的 dirty）
GER_yihui_bill_refresh = {
	add_to_variable = { GER_yihui_bill_dirty = 1 }
}

# 界面要用的数组先备好（没表决前也不能越界）
GER_yihui_bill_ensure_arrays = {
	if = { limit = { NOT = { check_variable = { GER_yihui_seat_bucket^num > 0 } } } GER_yihui_bill_init_buckets = yes }
	if = {
		limit = { check_variable = { GER_yihui_seat_code^num < 459 } }
		resize_array = { array = GER_yihui_seat_code value = 1 size = 459 }
	}
}

# 换一组「可用」（刚变可用的那组从头播）
GER_yihui_bill_flip = {
	if = { limit = { check_variable = { GER_yihui_bill_set = 0 } } set_variable = { GER_yihui_bill_set = 1 } }
	else = { set_variable = { GER_yihui_bill_set = 0 } }
	GER_yihui_bill_refresh = yes
}

# 表决完：按 GER_yihui_vote（每个座位的票）算好编码，换一组播
#   座位：帧 = 票 × 32 + 批次 + 1
#   计票条：帧 = 赞成千分比 + 反对千分比 × 1000 + 1
#   票数：帧 = 票数 + 边 × 1000 + 1（边 1 赞成 / 2 反对）
GER_yihui_bill_build = {
	GER_Reichstag_initialize = yes
	GER_yihui_bill_ensure_arrays = yes
	for_each_loop = {
		array = GER_yihui_vote
		value = yh_v
		index = yh_s
		set_temp_variable = { yh_f = yh_v }
		multiply_temp_variable = { yh_f = 32 }
		add_to_temp_variable = { yh_f = GER_yihui_seat_bucket^yh_s }
		add_to_temp_variable = { yh_f = 1 }
		set_variable = { GER_yihui_seat_code^yh_s = yh_f }
	}
	set_temp_variable = { yh_total = GER_yihui_vote^num }
	set_temp_variable = { yh_yp = GER_yihui_vote_yes }
	multiply_temp_variable = { yh_yp = 1000 }
	divide_temp_variable = { yh_yp = yh_total }
	round_temp_variable = yh_yp
	set_temp_variable = { yh_np = GER_yihui_vote_no }
	multiply_temp_variable = { yh_np = 1000 }
	divide_temp_variable = { yh_np = yh_total }
	round_temp_variable = yh_np
	set_variable = { GER_yihui_bill_bar_frame = yh_np }
	multiply_variable = { GER_yihui_bill_bar_frame = 1000 }
	add_to_variable = { GER_yihui_bill_bar_frame = yh_yp }
	add_to_variable = { GER_yihui_bill_bar_frame = 1 }
	set_variable = { GER_yihui_bill_yes_frame = GER_yihui_vote_yes }
	add_to_variable = { GER_yihui_bill_yes_frame = 1001 }
	set_variable = { GER_yihui_bill_no_frame = GER_yihui_vote_no }
	add_to_variable = { GER_yihui_bill_no_frame = 2001 }
	GER_yihui_bill_flip = yes
}

	set_country_flag = GER_yihui_bill_open
	GER_yihui_bill_refresh = yes

	clr_country_flag = GER_yihui_bill_open
	GER_yihui_bill_refresh = yes
}

# 右上角按钮（左栏和弹窗同一个位置）：开着就关；关着就开，并换一组让动画从头播
GER_yihui_bill_toggle = {
	if = {
		limit = { has_country_flag = GER_yihui_bill_open }
		GER_yihui_bill_close = yes
	}
	else = {
		GER_yihui_bill_flip = yes
		GER_yihui_bill_open = yes
	}
}

# 回到没表决时的默认状态（暗座位、空计票条）；表决结果到期也调这个
GER_yihui_bill_clear = {
	set_variable = { GER_yihui_bill_id = 0 }
	set_variable = { GER_yihui_bill_cost = 0 }
	set_variable = { GER_yihui_bill_days = 0 }
	GER_yihui_bill_refresh = yes
}

GER_yihui_bill_daily = {
	if = {
		limit = { check_variable = { GER_yihui_bill_days > 0 } }
		subtract_from_variable = { GER_yihui_bill_days = 1 }
		if = {
			limit = { check_variable = { GER_yihui_bill_days < 1 } }
			GER_yihui_bill_clear = yes
		}
	}
}
