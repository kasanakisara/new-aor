# The newspaper has 100 whole seats. Round only the display, preserving real fractional support.
GER_SPD_refresh_display_support_effect = {
    set_variable = { GER_SPD_left_display_support = GER_SPD_left_support }
    set_variable = { GER_SPD_center_display_support = GER_SPD_center_support }
    set_variable = { GER_SPD_right_display_support = GER_SPD_right_support }
    set_temp_variable = { GER_SPD_total = 100 }
	set_temp_variable = { GER_SPD_left_quota = GER_SPD_left_display_support }
	multiply_temp_variable = { GER_SPD_left_quota = 100 }
	divide_temp_variable = { GER_SPD_left_quota = GER_SPD_total }
	set_variable = { GER_SPD_left_display_support = GER_SPD_left_quota }
	round_variable = GER_SPD_left_display_support
	if = {
		limit = { check_variable = { GER_SPD_left_display_support > GER_SPD_left_quota } }
		add_to_variable = { GER_SPD_left_display_support = -1 }
	}
	set_temp_variable = { GER_SPD_left_remainder = GER_SPD_left_quota }
	subtract_from_temp_variable = { GER_SPD_left_remainder = GER_SPD_left_display_support }

	set_temp_variable = { GER_SPD_center_quota = GER_SPD_center_display_support }
	multiply_temp_variable = { GER_SPD_center_quota = 100 }
	divide_temp_variable = { GER_SPD_center_quota = GER_SPD_total }
	set_variable = { GER_SPD_center_display_support = GER_SPD_center_quota }
	round_variable = GER_SPD_center_display_support
	if = {
		limit = { check_variable = { GER_SPD_center_display_support > GER_SPD_center_quota } }
		add_to_variable = { GER_SPD_center_display_support = -1 }
	}
	set_temp_variable = { GER_SPD_center_remainder = GER_SPD_center_quota }
	subtract_from_temp_variable = { GER_SPD_center_remainder = GER_SPD_center_display_support }

	set_temp_variable = { GER_SPD_right_quota = GER_SPD_right_display_support }
	multiply_temp_variable = { GER_SPD_right_quota = 100 }
	divide_temp_variable = { GER_SPD_right_quota = GER_SPD_total }
	set_variable = { GER_SPD_right_display_support = GER_SPD_right_quota }
	round_variable = GER_SPD_right_display_support
	if = {
		limit = { check_variable = { GER_SPD_right_display_support > GER_SPD_right_quota } }
		add_to_variable = { GER_SPD_right_display_support = -1 }
	}
	set_temp_variable = { GER_SPD_right_remainder = GER_SPD_right_quota }
	subtract_from_temp_variable = { GER_SPD_right_remainder = GER_SPD_right_display_support }

	set_temp_variable = { GER_SPD_unassigned = 100 }
	subtract_from_temp_variable = { GER_SPD_unassigned = GER_SPD_left_display_support }
	subtract_from_temp_variable = { GER_SPD_unassigned = GER_SPD_center_display_support }
	subtract_from_temp_variable = { GER_SPD_unassigned = GER_SPD_right_display_support }
	while = {
		limit = { check_variable = { GER_SPD_unassigned > 0 } }
		if = {
			limit = {
				NOT = { check_variable = { GER_SPD_left_remainder < GER_SPD_center_remainder } }
				NOT = { check_variable = { GER_SPD_left_remainder < GER_SPD_right_remainder } }
			}
			add_to_variable = { GER_SPD_left_display_support = 1 }
			set_temp_variable = { GER_SPD_left_remainder = -1 }
		}
		else_if = {
			limit = { NOT = { check_variable = { GER_SPD_center_remainder < GER_SPD_right_remainder } } }
			add_to_variable = { GER_SPD_center_display_support = 1 }
			set_temp_variable = { GER_SPD_center_remainder = -1 }
		}
		else = {
			add_to_variable = { GER_SPD_right_display_support = 1 }
			set_temp_variable = { GER_SPD_right_remainder = -1 }
		}
		add_to_temp_variable = { GER_SPD_unassigned = -1 }
	}
}
