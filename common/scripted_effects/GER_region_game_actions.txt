# 地区博弈 · 行动方针（决议）的效果（手写；tools/GER_region_game/gen_region_game.py 只在末尾补缺的钩子，不改已有内容）

# 已由界面自动处理（不用在这里写）：扣政治点数、按天计时、生效中显示暗黄、驻区方针切换、每个地区的执行次数上限。
# 编号、名字、花费、天数、次数上限、只在哪个地区出现：gen_region_game.py 的 ACTIONS。
#
# 钩子（都在本国 GER 作用域执行；被操作的地区 = 变量 GER_region_target）：
#   GER_region_action_N            点击时（政治点数已扣）
#   GER_region_action_N_complete   计时结束那天（有生效天数的日常决议）
#   GER_region_action_N_biweekly   生效期间每满 14 天（生效 >= 14 天的日常决议、驻区方针）
#   GER_region_action_N_daily      生效期间每天（驻区方针）
# 小工具（先 set_temp_variable = { GER_region_delta = X } 或 { GER_region_value = X }，再调用；见 GER_region_game_effects.txt）：
#   GER_region_target_add_support / _add_control          目标地区支持度 / 控制度 + delta（夹在 0-100，控制度不低于下限）
#   GER_region_Rxx_add_support / _add_control             指定地区（R01 威悉-埃姆斯 …… R16 大柏林）
#   GER_region_target_neighbors_add_control               目标地区的相邻地区控制度 + delta
#   GER_region_target_add_control_growth                  目标地区永久「每两周控制度」+ delta
#   GER_region_target_set_state                           目标地区类型 = value（帧号：2 摇摆 3 左核 8 左控 ……）
#   GER_region_target_set_control_floor                   目标地区控制度下限 = value
# 注意：scripted_effect 不能是空的（实测空的引擎不认，点击会整段失效），每个钩子里至少留一句。
# 悬停介绍写在 localisation/simp_chinese/gui/GER_region_game_panel_l_simp_chinese.yml 的 GER_region_panel_row_N_tt，改了效果记得同步。

GER_region_bavaria_settle = {

        limit = { NOT = { has_country_flag = GER_bavaria_crisis_settled } }
        set_country_flag = GER_bavaria_crisis_settled
        set_temp_variable = { GER_region_bav_done = 0 }   # 一共执行了几条
        set_temp_variable = { GER_region_bav_four = 0 }   # 结局 2 的那四条里执行了几条
        if = { limit = { GER_region_bavaria_did_mediate = yes } add_to_temp_variable = { GER_region_bav_done = 1 } }
        if = { limit = { GER_region_bavaria_did_pressure = yes } add_to_temp_variable = { GER_region_bav_done = 1 } }
        if = { limit = { GER_region_bavaria_did_report = yes } add_to_temp_variable = { GER_region_bav_done = 1 } }
        if = { limit = { GER_region_bavaria_did_conservative = yes } add_to_temp_variable = { GER_region_bav_done = 1 } add_to_temp_variable = { GER_region_bav_four = 1 } }
        if = { limit = { GER_region_bavaria_did_denounce_terror = yes } add_to_temp_variable = { GER_region_bav_done = 1 } add_to_temp_variable = { GER_region_bav_four = 1 } }
        if = { limit = { GER_region_bavaria_did_withdraw = yes } add_to_temp_variable = { GER_region_bav_done = 1 } add_to_temp_variable = { GER_region_bav_four = 1 } }
        if = { limit = { GER_region_bavaria_did_ultimatum = yes } add_to_temp_variable = { GER_region_bav_done = 1 } add_to_temp_variable = { GER_region_bav_four = 1 } }
        if = { limit = { GER_region_bavaria_did_entente = yes } add_to_temp_variable = { GER_region_bav_done = 1 } }
        if = {
            limit = { has_country_flag = GER_bavaria_revolution_knock }
            set_variable = { GER_region_bavaria_result = 4 }
        }
        else_if = {
            limit = {
                GER_region_bavaria_did_mediate = yes
                GER_region_bavaria_did_denounce_terror = yes
                GER_region_bavaria_did_withdraw = yes
                GER_region_bavaria_did_ultimatum = yes
                GER_region_bavaria_did_entente = yes
                has_country_flag = GER_bavaria_paris_letter
            }
            set_variable = { GER_region_bavaria_result = 1 }
        }
        else_if = {
            limit = {
                check_variable = { GER_region_bav_four > 1 }
                GER_region_bavaria_did_entente = yes
                GER_region_bavaria_did_mediate = yes
                OR = {
                    GER_region_bavaria_did_ultimatum = yes
                    has_country_flag = GER_bavaria_paris_letter
                    has_country_flag = GER_bavaria_austria_shadow
                }
            }
            set_variable = { GER_region_bavaria_result = 2 }
        }
        else_if = {
            limit = {
                OR = {
                    check_variable = { GER_region_bav_done > 1 }
                    GER_region_bavaria_did_entente = yes
                    has_country_flag = GER_bavaria_paris_letter
                    has_country_flag = GER_bavaria_austria_shadow
                }
            }
            set_variable = { GER_region_bavaria_result = 3 }
        }
        else = {
            set_variable = { GER_region_bavaria_result = 4 }
        }
        GER_region_game_bavaria_end = yes   # 巴伐利亚决议全部隐藏，地块保留
        if = { limit = { check_variable = { GER_region_bavaria_result = 1 } } country_event = { id = AOR_GER_bavaria_crisis.18 } }
        else_if = { limit = { check_variable = { GER_region_bavaria_result = 2 } } country_event = { id = AOR_GER_bavaria_crisis.19 } }
        else_if = { limit = { check_variable = { GER_region_bavaria_result = 3 } } country_event = { id = AOR_GER_bavaria_crisis.20 } }
        else = { country_event = { id = AOR_GER_bavaria_crisis.21 } }
    }
}

# 四个结局共同的效果（结局事件的按钮里调用，调用后要跑 GER_region_game_sync_frames）：巴伐利亚转换为右派核心区（帧 4）。
# 核心区不显示数值；背地里的支持度 / 控制度和地图全开时的右派核心区一样设成 0 / 0（gen_region_game.py 的 HIDDEN_VALUES）。
GER_region_bavaria_turn_core = {
    set_variable = { GER_region_R09_state = 4 }
    set_variable = { GER_region_R09_support = 0 }
    set_variable = { GER_region_R09_control = 0 }
}

# 结局对周边地区的影响（结局事件的按钮里调用，调用后要跑 GER_region_game_sync_frames）。只做一次。
# 地图还没全开时周边地区还没开放：先不做，等 GER_region_game_open_map 设好各地区的开局数值后由 GER_region_on_open_map 补做。
#   2 冲突和平落幕  巴伐利亚周边地区（相邻关系见 gen_region_game.py 的 ADJ：图林根-萨克森 R08、巴登-符腾堡 R10、黑森 R11）支持度、控制度 -10
#   3 苦涩的和平    同上，-50
#   4 全面溃败      黑森（R11）转换为右派控制区（帧 9）
GER_region_bavaria_apply_result = {
    if = {
        limit = {
            has_country_flag = GER_region_map_opened
            has_country_flag = GER_bavaria_crisis_settled
            NOT = { has_country_flag = GER_region_bavaria_result_applied }
        }
        set_country_flag = GER_region_bavaria_result_applied
        set_temp_variable = { GER_region_delta = 0 }
        if = { limit = { check_variable = { GER_region_bavaria_result = 2 } } set_temp_variable = { GER_region_delta = -10 } }
        if = { limit = { check_variable = { GER_region_bavaria_result = 3 } } set_temp_variable = { GER_region_delta = -50 } }
        if = {
            limit = { check_variable = { GER_region_delta < 0 } }
            GER_region_R08_add_support = yes
            GER_region_R08_add_control = yes
            GER_region_R10_add_support = yes
            GER_region_R10_add_control = yes
            GER_region_R11_add_support = yes
            GER_region_R11_add_control = yes
        }
        if = {
            limit = { check_variable = { GER_region_bavaria_result = 4 } }
            set_variable = { GER_region_R11_state = 9 }
        }
    }
}

# 地图全开（GER_region_game_open_map）时，各地区开局数值设好之后调用
GER_region_on_open_map = {
    GER_region_bavaria_apply_result = yes
}

GER_region_actions_on_reset = {
    set_variable = { GER_region_strikes_active = 0 }
    if = {
        limit = { has_dynamic_modifier = { modifier = GER_region_strike_modifier } }
        remove_dynamic_modifier = { modifier = GER_region_strike_modifier }
    }
    if = {
        limit = { has_dynamic_modifier = { modifier = GER_region_bavaria_withdraw_modifier } }
        remove_dynamic_modifier = { modifier = GER_region_bavaria_withdraw_modifier }
    }
    clr_country_flag = GER_bavaria_crisis_settled
    clr_country_flag = GER_region_bavaria_result_applied
}
