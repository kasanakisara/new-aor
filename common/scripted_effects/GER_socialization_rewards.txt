# Parliamentary approval hook: call GER_socialization_enact_amendment = yes from the future vote event.
GER_socialization_enact_amendment = {
    set_country_flag = GER_socialization_amendment_passed
    GER_socialization_refresh_bars = yes
}

    set_variable = { GER_soc_target_metalworking = 700 }
    set_variable = { GER_soc_target_chemicals = 400 }
    set_variable = { GER_soc_target_electrical = 600 }
    set_variable = { GER_soc_target_handicrafts = 300 }

        limit = { has_country_flag = GER_socialization_amendment_passed }
    set_variable = { GER_soc_target_metalworking = 350 }
    set_variable = { GER_soc_target_chemicals = 200 }
    set_variable = { GER_soc_target_electrical = 350 }
    set_variable = { GER_soc_target_handicrafts = 200 }
    }
}

GER_socialization_award_railway = {
    if = {
        limit = { NOT = { has_country_flag = GER_socialization_reward_railway } }
        set_country_flag = GER_socialization_reward_railway
        subtract_from_variable = { GER_socialization_counterrevolution = 60 }
        clamp_variable = { var = GER_socialization_counterrevolution min = 0 max = 1200 }
    }
}

        set_country_flag = GER_socialization_reward_mining
        subtract_from_variable = { GER_socialization_counterrevolution = 60 }

# Resource-specific output modifiers are state scoped. Sync owned states after
# annexations and remove the bonus from states no longer owned by Germany.
GER_socialization_refresh_mining_resources = {
    every_state = {
        limit = {
            has_state_flag = GER_socialization_mining_resource_bonus
            OR = {
                NOT = { owner = { tag = GER } }
                NOT = { GER = { has_country_flag = GER_socialization_reward_mining } }
            }
        }
        remove_dynamic_modifier = { modifier = GER_socialization_mining_resource_state }
        clr_state_flag = GER_socialization_mining_resource_bonus
    }
    if = {
        limit = { has_country_flag = GER_socialization_reward_mining }
        every_owned_state = {
            limit = { NOT = { has_state_flag = GER_socialization_mining_resource_bonus } }
            add_dynamic_modifier = { modifier = GER_socialization_mining_resource_state }
            set_state_flag = GER_socialization_mining_resource_bonus
        }
    }
}

# One visible country modifier gathers the permanent bonuses unlocked by each
# industry. Keep the older ideas defined so saves can remove them on migration.
GER_socialization_refresh_industry_rewards = {
    remove_ideas = GER_socialization_reward_power_idea
    remove_ideas = GER_socialization_reward_armaments_idea
    remove_ideas = GER_socialization_reward_metalworking_idea
    remove_ideas = GER_socialization_reward_light_industry_idea
    remove_ideas = GER_socialization_reward_shipbuilding_idea
    set_variable = { GER_soc_reward_stability_weekly = 0 }
    set_variable = { GER_soc_reward_factory_output = 0 }
    set_variable = { GER_soc_reward_factory_efficiency = 0 }
    set_variable = { GER_soc_reward_construction_speed = 0 }
    set_variable = { GER_soc_reward_consumer_goods = 0 }
    set_variable = { GER_soc_reward_dockyard_output = 0 }
    if = { limit = { has_country_flag = GER_socialization_reward_power } set_variable = { GER_soc_reward_stability_weekly = 0.0035 } }
    if = { limit = { has_country_flag = GER_socialization_reward_armaments } set_variable = { GER_soc_reward_factory_output = 0.10 } set_variable = { GER_soc_reward_factory_efficiency = 0.10 } }
    if = { limit = { has_country_flag = GER_socialization_reward_metalworking } set_variable = { GER_soc_reward_construction_speed = 0.10 } }
    if = { limit = { has_country_flag = GER_socialization_reward_light_industry } set_variable = { GER_soc_reward_consumer_goods = -0.10 } }
    if = { limit = { has_country_flag = GER_socialization_reward_shipbuilding } set_variable = { GER_soc_reward_dockyard_output = 0.10 } }
    if = {
        limit = {
            OR = {
                has_country_flag = GER_socialization_reward_railway
                has_country_flag = GER_socialization_reward_mining
                has_country_flag = GER_socialization_reward_power
                has_country_flag = GER_socialization_reward_banking
                has_country_flag = GER_socialization_reward_armaments
                has_country_flag = GER_socialization_reward_metalworking
                has_country_flag = GER_socialization_reward_chemicals
                has_country_flag = GER_socialization_reward_electrical
                has_country_flag = GER_socialization_reward_light_industry
                has_country_flag = GER_socialization_reward_handicrafts
                has_country_flag = GER_socialization_reward_shipbuilding
            }
            NOT = { has_dynamic_modifier = { modifier = GER_socialization_industry_rewards_modifier } }
        }
        add_dynamic_modifier = { modifier = GER_socialization_industry_rewards_modifier }
    }
}
