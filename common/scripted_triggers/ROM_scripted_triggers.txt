is_available_fighter_ROM = {
	OR = {
		AND = {
			has_tech = aor_air_small_1919
			ROM = { 
				NOT = { has_tech = aor_air_small_1919}
			}
		}
		AND = {
			has_tech = aor_air_small_1925
			ROM = { 
				NOT = { has_tech = aor_air_small_1925}
			}
		}
		AND = {
			has_tech = aor_air_small_1930
			ROM = { 
				NOT = { has_tech = aor_air_small_1930}
			}
		}
	}
	NOT = {
		has_war_with = ROM
	}
}
